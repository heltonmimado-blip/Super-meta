function falarTexto(texto) { if ('speechSynthesis' in window) { // Para qualquer leitura anterior para não sobrepor window.speechSynthesis.cancel(); const utterance = new SpeechSynthesisUtterance(texto); utterance.lang = 'pt-PT'; // Podes usar 'pt-BR' para sotaque do Brasil // Podes ajustar a velocidade (1 é normal) e o tom (1 é normal) utterance.rate = 1.0; utterance.pitch = 1.0; window.speechSynthesis.speak(utterance); } else { alert("O teu navegador não suporta leitura de voz."); } }
