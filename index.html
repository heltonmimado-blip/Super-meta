const SCHOOL_BOOK_SOURCES = [ { name:"Ministério da Educação de Moçambique", url:"https://www.mined.gov.mz/" }, { name:"IEDA - Instituto de Educação Aberta e à Distância", url:"https://ieda.gov.mz/" } ]; function searchSchoolBooks(){ const level=document.getElementById("libraryLevel").value; const subject=document.getElementById("librarySubject").value; const container=document.getElementById("onlineBooks"); container.replaceChildren(); const title=document.createElement("p"); title.textContent="Pesquisa de manuais: "+level+" · "+subject; container.appendChild(title); SCHOOL_BOOK_SOURCES.forEach(source=>{ const card=document.createElement("div"); card.className="book"; const name=document.createElement("strong"); name.textContent=source.name; const description=document.createElement("p"); description.className="small"; description.textContent="Consultar materiais escolares e procurar PDFs disponíveis."; const link=document.createElement("a"); link.href=source.url; link.target="_blank"; link.rel="noopener noreferrer"; link.textContent="Abrir biblioteca oficial ↗"; link.style.color="#75aaff"; card.append(name,description,link); container.appendChild(card); }); const search=document.createElement("button"); search.className="btn secondary full"; search.textContent="Pesquisar na Internet"; search.onclick=()=>{ const query=encodeURIComponent( "livro escolar Moçambique PDF "+level+" "+subject ); window.open( "https://www.google.com/search?q="+query, "_blank" ); }; container.appendChild(search); }
